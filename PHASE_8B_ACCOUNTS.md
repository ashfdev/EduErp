--- Chart of Accounts ---

model AccountGroup {
  id            String        @id @default(cuid())
  name          String        // "Assets", "Liabilities", "Income", "Expenses"
  type          AccountType
  display_order Int
  created_at    DateTime      @default(now())
  accounts      Account[]
}

model Account {
  id              String        @id @default(cuid())
  account_group_id String
  account_group   AccountGroup  @relation(fields: [account_group_id], references: [id])
  code            String        @unique  // e.g. "1001", "2001", "4001"
  name            String        // "Cash in Hand", "Tuition Fee Income"
  name_bn         String?
  account_nature  AccountNature // DEBIT_NORMAL or CREDIT_NORMAL
  is_bank_account Boolean       @default(false)
  is_cash_account Boolean       @default(false)
  is_system       Boolean       @default(false)  // auto-created, cannot delete
  opening_balance Float         @default(0)
  opening_balance_type BalanceType @default(DEBIT)
  is_active       Boolean       @default(true)
  created_at      DateTime      @default(now())
  updated_at      DateTime      @updatedAt
  debit_entries   JournalEntry[] @relation("DebitAccount")
  credit_entries  JournalEntry[] @relation("CreditAccount")
  bank_account    BankAccount?
  budget_lines    BudgetLine[]
  @@index([account_group_id])
}

model BankAccount {
  id              String    @id @default(cuid())
  account_id      String    @unique
  account         Account   @relation(fields: [account_id], references: [id])
  bank_name       String    // "Sonali Bank", "Dutch Bangla Bank"
  branch_name     String?
  account_number  String
  account_title   String
  current_balance Float     @default(0)
  created_at      DateTime  @default(now())
  reconciliations BankReconciliation[]
  cheques         ChequeEntry[]
}

--- Vouchers ---

model Voucher {
  id                String        @id @default(cuid())
  voucher_no        String        @unique  // RV-2026-0001, PV-2026-0001, JV-2026-0001
  voucher_type      VoucherType
  financial_year_id String
  financial_year    FinancialYear @relation(fields: [financial_year_id], references: [id])
  date              DateTime      @db.Date
  narration         String        @db.Text  // description of the transaction
  narration_bn      String?       @db.Text
  reference_no      String?       // cheque no, invoice no, etc.
  reference_type    String?       // "FEE_PAYMENT", "PAYROLL", "INVENTORY_PURCHASE"
  reference_id      String?       // FK to the originating record
  total_amount      Float
  status            VoucherStatus @default(DRAFT)
  created_by_id     String
  approved_by_id    String?
  approved_at       DateTime?
  is_auto           Boolean       @default(false)  // auto-generated by system
  attachment_url    String?
  deleted_at        DateTime?
  created_at        DateTime      @default(now())
  updated_at        DateTime      @updatedAt
  journal_entries   JournalEntry[]
  @@index([date])
  @@index([voucher_type])
  @@index([financial_year_id])
}

model JournalEntry {
  id              String    @id @default(cuid())
  voucher_id      String
  voucher         Voucher   @relation(fields: [voucher_id], references: [id])
  debit_account_id  String?
  debit_account     Account? @relation("DebitAccount", fields: [debit_account_id], references: [id])
  credit_account_id String?
  credit_account    Account? @relation("CreditAccount", fields: [credit_account_id], references: [id])
  amount          Float
  narration       String?   // line-level narration
  created_at      DateTime  @default(now())
  @@index([voucher_id])
  @@index([debit_account_id])
  @@index([credit_account_id])
}

--- Bank Reconciliation ---

model BankReconciliation {
  id                String      @id @default(cuid())
  bank_account_id   String
  bank_account      BankAccount @relation(fields: [bank_account_id], references: [id])
  month             Int
  year              Int
  statement_balance Float       // from bank statement
  book_balance      Float       // from ledger
  difference        Float
  status            ReconcileStatus @default(PENDING)
  reconciled_at     DateTime?
  notes             String?
  items             ReconciliationItem[]
  @@unique([bank_account_id, month, year])
}

model ReconciliationItem {
  id                    String              @id @default(cuid())
  reconciliation_id     String

  voucher_id            String?

  is_reconciled         Boolean             @default(false)
}

--- Financial Year Endpoints ---

GET  /api/accounts/financial-years
POST /api/accounts/financial-years          → create year
PUT  /api/accounts/financial-years/:id/activate → set active (deactivate others)
POST /api/accounts/financial-years/:id/close
  Body: { confirmed: true }
  Logic:
    1. Calculate total income vs total expense for the year
    2. Create closing journal entry: Surplus/Deficit = Income - Expense
    3. Set FinancialYear.is_closed = true
    4. Create next year's opening balances (carry forward balance sheet accounts)
  Returns: closing summary + surplus/deficit amount

--- Voucher Endpoints ---

GET  /api/accounts/vouchers
  Query: financial_year_id, type, status, from_date, to_date, search, page, limit
  Returns: paginated list with total_amount per voucher

GET  /api/accounts/vouchers/:id
  Returns: full voucher with all journal_entries and account names

POST /api/accounts/vouchers
  Body: {
    voucher_type: "RECEIPT"|"PAYMENT"|"JOURNAL"|"CONTRA",
    date: "2026-07-01",
    narration: "...",
    narration_bn: "...",
    reference_no?: "CHQ-001",
    entries: [
      { debit_account_id: "...", credit_account_id: "...", amount: 5000, narration?: "..." }
    ]
  }
  Validation:
    - For each entry: either debit_account_id or credit_account_id must be set (or both)
    - Total debits must equal total credits (double-entry balance rule)
    - Sum check: debit_total === credit_total else 400 "Voucher is unbalanced"
    - Date must be within active FinancialYear date range
    - Active FinancialYear must not be closed
  
  Auto-generate voucher_no:
    RV-{YYYY}-{seq} for RECEIPT
    PV-{YYYY}-{seq} for PAYMENT
    JV-{YYYY}-{seq} for JOURNAL
    CV-{YYYY}-{seq} for CONTRA
  
  Status: DRAFT on creation

PUT  /api/accounts/vouchers/:id
  Only if status=DRAFT
  Recalculate balance validation

POST /api/accounts/vouchers/:id/approve
  Authorized: ACCOUNTANT, ADMIN, PRINCIPAL
  Status: DRAFT → APPROVED
  Update account running balances in Redis cache

POST /api/accounts/vouchers/:id/post
  Authorized: ACCOUNTANT, ADMIN
  Status: APPROVED → POSTED
  Once POSTED: cannot be edited or deleted (only reversed via new voucher)

POST /api/accounts/vouchers/:id/cancel
  Only if status=DRAFT or APPROVED (not POSTED)
  Status → CANCELLED

async function createFeeReceiptJournal(payment: Payment, invoice: Invoice): Promise<Voucher>
  Journal: Dr Cash/Bank → Cr [Fee Income Account based on category]
  
  Fee category → Income account mapping (configurable in settings):
    TUITION  → 4001 Tuition Fee Income
    ADMISSION → 4002 Admission Fee Income
    EXAM     → 4003 Exam Fee Income
    TRANSPORT → 4004 Transport Fee Income
    HOSTEL   → 4005 Hostel Fee Income
    LAB      → 4006 Lab Fee Income
    LIBRARY  → 4007 Library Fee Income
    LATE_FINE → 4009 Late Fine Income
  
  If payment via bKash/Nagad/SSLCommerz: Dr 1002 (Bank) → Cr Income
  If payment CASH: Dr 1001 (Cash in Hand) → Cr Income
  
  Set: is_auto=true, reference_type="FEE_PAYMENT", reference_id=payment.id
  Auto-post the voucher (no approval needed for auto entries)

async function createPayrollJournal(payrollRecord: PayrollRecord, staff: Staff): Promise<Voucher>
  Journal:
    Dr 5001/5002 Salary Expense → Cr 2004 Salary Payable (gross amount)
    Dr 2004 Salary Payable → Cr 1001/1002 Cash/Bank (net amount paid)
    Dr 2004 Salary Payable → Cr 2002 TDS Payable (TDS deducted)
    Dr 5004 PF Expense → Cr 2??? PF Payable (employer PF contribution)

async function createInventoryPurchaseJournal(purchase: InventoryPurchase): Promise<Voucher>
  Journal: Dr [Asset/Expense Account] → Cr 2001 Accounts Payable OR Cr 1001/1002 Cash/Bank

async function createDepreciationJournal(period: string, entries: DepreciationEntry[]): Promise<Voucher>
  Journal: Dr 5018 Depreciation Expense → Cr 1105 Accumulated Depreciation
